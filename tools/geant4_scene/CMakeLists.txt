cmake_minimum_required(VERSION 3.28)
project(geant4_scene LANGUAGES CXX)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
file(REAL_PATH "${CMAKE_BINARY_DIR}" BUILD_REAL)
file(RELATIVE_PATH BUILD_REL "${PROJECT_ROOT}/.local/geometry-events-publication/build" "${BUILD_REAL}")
if(BUILD_REL MATCHES "^\\.\\." OR IS_ABSOLUTE "${BUILD_REL}")
  message(FATAL_ERROR "Build only under .local/geometry-events-publication/build")
endif()
find_package(Geant4 11.3.2 EXACT REQUIRED gdml)
include(${Geant4_USE_FILE})
add_executable(geant4_scene scene.cc)
target_compile_features(geant4_scene PRIVATE cxx_std_17)
target_link_libraries(geant4_scene PRIVATE ${Geant4_LIBRARIES})
target_compile_definitions(geant4_scene PRIVATE PROJECT_ROOT="${PROJECT_ROOT}")
