cmake_minimum_required(VERSION 3.28)
project(gegi_scene LANGUAGES CXX)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
file(REAL_PATH "${CMAKE_BINARY_DIR}" BUILD_REAL)
file(RELATIVE_PATH BUILD_REL "${PROJECT_ROOT}/.local/gegi-website/build" "${BUILD_REAL}")
if(BUILD_REL MATCHES "^\\.\\." OR IS_ABSOLUTE "${BUILD_REL}")
  message(FATAL_ERROR "Build only under .local/gegi-website/build")
endif()
find_package(Geant4 11.3.2 EXACT REQUIRED gdml)
include(${Geant4_USE_FILE})
add_executable(gegi_scene ../gegi_scene.cc)
target_compile_features(gegi_scene PRIVATE cxx_std_17)
target_link_libraries(gegi_scene PRIVATE ${Geant4_LIBRARIES})
target_compile_definitions(gegi_scene PRIVATE PROJECT_ROOT="${PROJECT_ROOT}")
