cmake_minimum_required(VERSION 3.28)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(RELATIVE_PATH BUILD_RELATIVE "${PROJECT_ROOT}/.local/m2a" "${CMAKE_BINARY_DIR}")
if(BUILD_RELATIVE MATCHES "^\\.\\." OR IS_ABSOLUTE "${BUILD_RELATIVE}")
  message(FATAL_ERROR "Build under project .local/m2a, never in the source directory")
endif()
project(catalog_geometry_export LANGUAGES CXX)
find_package(Geant4 REQUIRED gdml)
include(${Geant4_USE_FILE})
add_executable(cryostat_catalog_export ../cryostat_catalog_export.cc)
target_compile_features(cryostat_catalog_export PRIVATE cxx_std_17)
target_link_libraries(cryostat_catalog_export PRIVATE ${Geant4_LIBRARIES})
target_compile_definitions(cryostat_catalog_export PRIVATE PROJECT_LOCAL="${PROJECT_ROOT}/.local")
