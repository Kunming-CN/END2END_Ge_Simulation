cmake_minimum_required(VERSION 3.28)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(RELATIVE_PATH BUILD_RELATIVE "${PROJECT_ROOT}/.local/m2a" "${CMAKE_BINARY_DIR}")
if(BUILD_RELATIVE MATCHES "^\\.\\." OR IS_ABSOLUTE "${BUILD_RELATIVE}")
  message(FATAL_ERROR "Build under project .local/m2a, never in the source directory")
endif()
project(gps_command_precision_test LANGUAGES CXX)
find_package(Geant4 11.3.2 EXACT REQUIRED)
include(${Geant4_USE_FILE})
add_executable(test_gps_precision ../test_gps_precision.cc)
target_compile_features(test_gps_precision PRIVATE cxx_std_17)
target_link_libraries(test_gps_precision PRIVATE ${Geant4_LIBRARIES})
